function salveaza() { let nume = document.getElementById("nume").value.trim(); let telefon = document.getElementById("telefon").value.trim(); let buget = document.getElementById("buget").value.trim(); // validare câmpuri if (!nume || !telefon || !buget) { alert("Completează toate câmpurile!"); return; } // transformare număr în format internațional (pentru România) if (telefon.startsWith("0")) { telefon = "4" + telefon; // ex: 0712345678 -> 40712345678 } // creează lead-ul let lead = `${nume} - ${telefon} - ${buget}`; let lista = document.getElementById("lista"); let li = document.createElement("li"); // creează text let text = document.createElement("span"); text.textContent = lead + " "; // creează buton WhatsApp let btn = document.createElement("button"); btn.textContent = "WhatsApp"; // stilizare buton btn.style.backgroundColor = "#25D366"; btn.style.color = "white"; btn.style.border = "none"; btn.style.padding = "5px 10px"; btn.style.cursor = "pointer"; btn.style.borderRadius = "5px"; // mesaj WhatsApp let mesaj = `Buna ${nume}! Te contactez de la CIOBANU Real Estate pentru proprietatea dorita.`; btn.onclick = function() { let url = "https://wa.me/" + telefon + "?text=" + encodeURIComponent(mesaj); window.open(url, "_blank"); // deschide în fereastră nouă }; li.appendChild(text); li.appendChild(btn); lista.appendChild(li); // curăță formularul document.getElementById("nume").value = ""; document.getElementById("telefon").value = ""; document.getElementById("buget").value = ""; }
